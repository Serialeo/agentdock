cmake_minimum_required(VERSION 3.20)
project(AgentDockComputerSpike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(computer-geometry-test geometry_test.cpp)
# Keep assertions active in Release builds as well.
if(MSVC)
  target_compile_options(computer-geometry-test PRIVATE /UNDEBUG)
else()
  target_compile_options(computer-geometry-test PRIVATE -UNDEBUG)
endif()
add_test(NAME computer-geometry COMMAND computer-geometry-test)
if(WIN32)
  add_executable(AgentDockComputerSpike WIN32 main.cpp)
  target_compile_definitions(AgentDockComputerSpike PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(AgentDockComputerSpike PRIVATE d3d11 dxgi windowscodecs ole32 user32 gdi32 advapi32 shell32 uuid)
  if(MSVC)
    target_compile_options(AgentDockComputerSpike PRIVATE /W4 /utf-8)
    target_link_options(AgentDockComputerSpike PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
  else()
    message(FATAL_ERROR "Native spike requires the MSVC Windows SDK toolchain; portable tests can run on Linux.")
  endif()
endif()
